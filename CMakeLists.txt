cmake_minimum_required(VERSION 3.20)
project(poker_gto_solver LANGUAGES CXX)

option(POKER_ENABLE_CUDA "Build CUDA backend" OFF)
option(POKER_BUILD_PYTHON "Build pybind11 module when pybind11 is available" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(poker_core solver/src/game.cpp solver/src/cfr.cpp)
target_include_directories(poker_core PUBLIC solver/include)
target_compile_options(poker_core PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(poker_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(POKER_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(poker_cuda solver/cuda/cfr_kernels.cu)
  target_include_directories(poker_cuda PUBLIC solver/include)
  target_link_libraries(poker_cuda PUBLIC poker_core CUDA::cudart)
  set_target_properties(poker_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
else()
  add_library(poker_cuda solver/src/cuda_solver_stub.cpp)
  target_include_directories(poker_cuda PUBLIC solver/include)
  target_link_libraries(poker_cuda PUBLIC poker_core)
endif()

add_executable(poker_solver solver/src/main.cpp)
target_link_libraries(poker_solver PRIVATE poker_core poker_cuda)
add_executable(benchmark solver/src/benchmark.cpp)
target_link_libraries(benchmark PRIVATE poker_core poker_cuda)

enable_testing()
add_executable(solver_tests solver/tests/test_solver.cpp)
target_link_libraries(solver_tests PRIVATE poker_core)
add_test(NAME solver_tests COMMAND solver_tests)

if(POKER_BUILD_PYTHON)
  find_package(pybind11 CONFIG QUIET)
  if(pybind11_FOUND)
    pybind11_add_module(_poker_solver solver/src/bindings.cpp)
    target_link_libraries(_poker_solver PRIVATE poker_core poker_cuda)
    install(TARGETS _poker_solver LIBRARY DESTINATION poker_solver)
  else()
    message(STATUS "pybind11 not found; native Python module will not be built")
  endif()
endif()
